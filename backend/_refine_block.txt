        ], { maxTokens: 1500 });

          title: d.title || '未命名方向',
          statement: d.statement || '',
          questions: Array.isArray(d.questions) ? d.questions : [],
          gap: d.gap || '',
          papers: Array.isArray(d.papers) ? d.papers : [],
        });

        setRefineError(e instanceof Error ? e.message : String(e));
      }
      setPhase('done');
      return;
